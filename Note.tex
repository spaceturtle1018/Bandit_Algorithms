\section{Bandits, Probability and Concentration}
\subsection{Introduction}

\begin{definition*}{}
    \leavevmode
    \begin{itemize}
        \item Actions : ${A_1, A_2, \cdots } \in \mathcal{A}$ and we call $\mathcal{A}$ the action set.
        \item Rewards : ${X_1, X_2, \cdots } \in \mathcal{R}$.
        \item History : $H_{t-1} = (A_1, X_1, \cdots, A_{t-1}, X_{t-1})$.
        \item Policy : $\pi \in \Pi : H_{t-1} \mapsto A_t$ and we call $\Pi$ the competitor class.
        \item Environment: $\mathcal{E} : A_t \mapsto X_t$.
        \item Regret of a learner relative to a policy $\pi$ until $n \in \mathbb{N}$ is 
        \begin{equation*}
            R_n = \mathbb{E}\left[ \sum_{t=1}^{n} \mathcal{E}(\pi (H_{t-1})) - \sum_{t=1}^{n} X_t \right].
        \end{equation*}
        where $H_0$ represents the initial condition.
    \end{itemize}
\end{definition*}

\begin{example*}
    An environment is called a \textbf{stochastic Bernoulli bandit} if 
    \begin{itemize}
        \item $\mathcal{A} = \left \{ 1, 2, \ldots, k \right \}$,
        \item $X_t \in \left\{0, 1 \right\}$,
        \item $^\forall \hspace{0.1em} a \in \mathcal{A}$, $\exists \hspace{0.2em} \mu_a \in [0,1]^k \text{ such that } \mathbb{P} \left[ X_t = 1| A_t = a \right] = \mu_a$.
    \end{itemize}
    Here the optimal policy is to play fixed action $a^* = \argmax\limits_{a \in \mathcal{A}} \mu_a$.
    That is, the competitor class $\Pi$ is the set of all constant policies $\Pi = \left \{ \pi_1, \ldots , \pi_k \right \}$ where $\pi_i$ chooses action $i$ in every round.
    The regret relative to the optimal policy $\pi^*$ until n rounds is
    \begin{equation*}
        \begin{aligned}
            R_n &= \mathbb{E}\left[ \sum_{t=1}^{n} \mathcal{E}(\pi^* (H_{t-1})) - \sum_{t=1}^{n} X_t \right] \\
            &= n \max\limits_{a \in \mathcal{A}} \mu_a - \mathbb{E}\left[ \sum_{t=1}^{n} X_t \right].
        \end{aligned}
    \end{equation*}
\end{example*}

\begin{definition*}{}
    \textbf{Stochastic stationary bandits} are the environments restricted to generate the rewards in response to each action from a distribution independent of the previous history.
    \begin{example*}{Linear Stochastic Bandits}
        \leavevmode
        \begin{itemize}
            \item $\mathcal{A} \subset \mathbb{R}^d$,
            \item $X_t = \langle a, \theta \rangle + \eta_t$ for $\theta \in \mathbb{R}^d, a \in \mathcal{A}$ and $\eta_t$ is a standard Gaussian.
        \end{itemize}
    \end{example*}
\end{definition*}

\subsection{Foundations of Probability}

\begin{definition*}{}Let an \textbf{event set} $\Omega$ be given.
    \leavevmode
    \begin{itemize}
        \item A set $\mathcal{F} \subset 2^{\Omega}$ is called a \textbf{$\sigma$-algebra} on $\Omega$ if
        \begin{itemize}
            \item[(1)] $\Omega \in \mathcal{F}$,
            \item[(2)] $A \in \mathcal{F} \implies A^c \in \mathcal{F}$,
            \item[(3)] $A_1, A_2, \ldots \in \mathcal{F} \implies \bigcup_{i=1}^{\infty} A_i \in \mathcal{F}$.
        \end{itemize}
        \item $\mathcal{G} \subseteq \mathcal{F}$ is a \textbf{sub-$\sigma$-algebra} if $\mathcal{G}$ is a $\sigma$-algebra on $\Omega$.
        \item ($\Omega, \mathcal{F}$) is called a \textbf{measurable space} if $\mathcal{F}$ is a $\sigma$-algebra on $\Omega$.
        \item $\mathcal{B}(\mathbb{R})$ is the \textbf{Borel $\sigma$-algebra} on $\mathbb{R}$, which is the smallest $\sigma$-algebra containing all open intervals $(a, b)$ for $a, b \in \mathbb{R}$.
    \end{itemize}
\end{definition*}

\begin{definition*}{}Let a measurable space $(\Omega, \mathcal{F})$ be given.
    \leavevmode
    \begin{itemize}
        \item A function $\mu : \mathcal{F} \mapsto [0, \infty]$ is \textbf{$\sigma$-additive} if 
        \begin{itemize}
            \item[(1)] $\mu(\emptyset) = 0$,
            \item[(2)] $A_1, A_2, \ldots \in \mathcal{F}$ are disjoint $\implies \mu\left( \bigcup_{i=1}^{\infty} A_i \right) = \sum_{i=1}^{\infty} \mu(A_i)$.
        \end{itemize}
        \item A function $\mu : \mathcal{F} \mapsto [0, \infty]$ is a \textbf{measure} if $\mu$ is $\sigma$-additive.
        \item $(\Omega, \mathcal{F}, \mu)$ is called a \textbf{measure space} if $\mu$ is a measure on $(\Omega, \mathcal{F})$.
        \item A measure $\mu$ is a \textbf{probability measure} if $\mu(\Omega) = 1$.
        \item $(\Omega, \mathcal{F}, \mu)$ is called a \textbf{probability space} if $\mu$ is a probability measure on $(\Omega, \mathcal{F})$.
        \item A function $X : \Omega \mapsto \mathbb{R}$ is called a \textbf{measurable function} if $^\forall \hspace{0.1em} B \in \mathcal{B}(\mathbb{R})$, $X^{-1}(B) \in \mathcal{F}$.
        \item A \textbf{filtration} is a sequence $(\mathcal{F}_t)_{t=0}^n$ of sub-$\sigma$-algebras of $\mathcal{F}$ where $\mathcal{F}_t \subseteq \mathcal{F}_{t+1}$ for all $t<n$.
    \end{itemize}
\end{definition*}

\begin{definition*}{}Let a measure space $(\Omega, \mathcal{F}, \mu)$ be given.
    \leavevmode
    \begin{itemize}
        \item A function $f : \Omega \mapsto [0, \infty]$ is called a \textbf{simple function} if $f = \sum_{i=1}^{n} a_i \mathbbm{1}_{A_i}$ for some $a_1, a_2, \ldots, a_n \in [0, \infty]$ and $A_1, A_2, \ldots, A_n \in \mathcal{F}$.
        \item The \textbf{integral of a simple function} $f = \sum_{i=1}^{n} a_i \mathbbm{1}_{A_i}$ is defined as $\int_{\Omega} f d\mu := \sum_{i=1}^{n} a_i \mu(A_i)$.
        \item The \textbf{integral of a non-negative measurable function} $f : \Omega \mapsto [0, \infty]$ is defined as $\int_{\Omega} f d\mu := \sup\limits_{g} \int_{\Omega} g d\mu$ where the supremum is taken over all simple functions $g$ such that $0 \leq g(\omega) \leq f(\omega)$ for all $\omega \in \Omega$.
        \item The \textbf{integral of a measurable function} $f : \Omega \mapsto \mathbb{R}$ is defined as $\int_{\Omega} f d\mu := \int_{\Omega} f^+ d\mu - \int_{\Omega} f^- d\mu$ where $f^+ = \max(f, 0)$ and $f^- = \max(-f, 0)$.
        \item A function $f : \Omega \mapsto \mathbb{R}$ is called \textbf{integrable} if $\int_{\Omega} |f| d\mu < \infty$.
    \end{itemize}
\end{definition*}

\begin{definition*}{}Let a probability space $(\Omega, \mathcal{F}, \mathbb{P})$ be given.
    \leavevmode
    \begin{itemize}
        \item A measurable function $X : \Omega \mapsto \mathbb{R}$ is called a \textbf{random variable}.
        \item The \textbf{conditional probability} $\mathbb{P}[A|B] := \frac{\mathbb{P}[A \cap B]}{\mathbb{P}[B]}$ for $A, B \in \mathcal{F}$ and $\mathbb{P}[B] > 0$.
        \item Two events $A, B \in \mathcal{F}$ are called \textbf{independent} if $\mathbb{P}[A \cap B] = \mathbb{P}[A] \cdot \mathbb{P}[B]$.
        \item Two random variables $X, Y$ are called \textbf{independent} if $^\forall \hspace{0.1em} A, B \in \mathcal{B}(\mathbb{R})$, $\mathbb{P}[X \in A, Y \in B] = \mathbb{P}[X \in A] \cdot \mathbb{P}[Y \in B]$.
    \end{itemize}
\end{definition*}

\begin{definition*}{}Let a probability space $(\Omega, \mathcal{F}, \mathbb{P})$ and a random variable $X$ be given.
    \leavevmode
    \begin{itemize}
        \item The \textbf{expectation} of a random variable $X$ is defined as $\mathbb{E}[X] := \int_{\Omega} X d\mathbb{P}$.
        \item For a sub-$\sigma$-algebra $\mathcal{H} \subseteq \mathcal{F}$, the \textbf{conditional expectation} of $X$ given $\mathcal{H}$ is denoted by $\mathbb{E}[X|\mathcal{H}]$ 
        and is defined as a random variable $g$ on $\Omega$ such that for all $H \in \mathcal{H}$,
        \begin{equation*}
            \int_{H} g \hspace{0.2em} d\mathbb{P} = \int_{H} X d\mathbb{P}.
        \end{equation*}
    \end{itemize}
\end{definition*}

\begin{theorem*}{2.11}
    If $(\Omega, \mathcal{F}, \mathbb{P})$ is a probability space and $X$ is a integrable random variable and $\mathcal{H} \subseteq \mathcal{F}$ is a sub-$\sigma$-algebra, there exists a unique (almost surely) random variable $\mathbb{E}[X|\mathcal{H}]$ such that for all $H \in \mathcal{H}$,
    \begin{equation*}
        \int_{H} \mathbb{E}[X|\mathcal{H}] \hspace{0.2em} d\mathbb{P} = \int_{H} X d\mathbb{P}.
    \end{equation*}
\end{theorem*}

\begin{theorem*}{2.12}
    Let $(\Omega, \mathcal{F}, \mathbb{P})$ be a probability space, 
    $\mathcal{G}, \mathcal{G}_1, \mathcal{G}_2 \subseteq \mathcal{F}$ be sub-$\sigma$-algebras of $\mathcal{F}$,
    $X, Y$ be integrable random variables on $(\Omega, \mathcal{F}, \mathbb{P})$.
    Then the following properties hold:
    \begin{itemize}
        \item[\text{(1)}] If $X \geq 0$ almost surely, then $\mathbb{E}[X|\mathcal{G}] \geq 0$ almost surely.
        \item[\text{(2)}] $\mathbb{E}[1|\mathcal{G}] = 1$ almost surely.
        \item[\text{(3)}] $\mathbb{E}[X + Y|\mathcal{G}] = \mathbb{E}[X|\mathcal{G}] + \mathbb{E}[Y|\mathcal{G}]$ almost surely.
        \item[\text{(4)}] If $Y$ is $\mathcal{G}$-measurable and $\mathbb{E}[XY] < \infty$, then $\mathbb{E}[XY|\mathcal{G}] = Y \mathbb{E}[X|\mathcal{G}]$ almost surely.
        \item[\text{(5)}] If $\mathcal{G}_1 \subseteq \mathcal{G}_2$, then $\mathbb{E}[X|\mathcal{G}_1] = \mathbb{E}[\mathbb{E}[X|\mathcal{G}_2]|\mathcal{G}_1]$ almost surely.
    \end{itemize}
\end{theorem*}

\subsection{Stochastic Processes and Markov Chains}

\begin{definition*}{}
    Let two measurable spaces $(\mathcal{X}, \mathcal{F})$ and $(\mathcal{Y}, \mathcal{G})$ be given.
    \leavevmode
    \begin{itemize}
        \item $(\mathcal{X}, \mathcal{F})$ and $(\mathcal{Y}, \mathcal{G})$ are called \textbf{isomorphic} if there exists a bijective function $f : \mathcal{X} \mapsto \mathcal{Y}$ such that $A \in \mathcal{F} \iff f(A) \in \mathcal{G}$.
        \item $(\mathcal{X}, \mathcal{F})$ is a \textbf{Borel space} if it is isomorphic to $(\mathbb{R}, \mathcal{B}(A))$ for some set $A \in \mathcal{B}(\mathbb{R})$.
    \end{itemize}
\end{definition*}

\begin{theorem*}{3.1}
    Let $\mu$ be a probability measure on a Borel space $S$ and $\lambda$ be the Lebesgue measure on $([0,1], \mathcal{B}([0,1]))$.
    Then there exists a sequence of independent random element $X_1, X_2, \ldots$ on $([0,1], \mathcal{B}([0,1]), \lambda)$ such that $\lambda_{X_t}(\cdot) := \lambda(X^{-1}_t(\cdot)) = \mu(\cdot)$ for all $t$.
\end{theorem*}

\begin{definition*}{}
    For "countable" set $\mathcal{T}$, A stochastic process is a collection of random variables ${X_t : t \in \mathcal{T}}$.
\end{definition*}

\begin{theorem*}{3.2}
    Let Borel spaces $(\Omega_n, \mathcal{F}_n)$ and measures $\mu_n$ on $(\Omega_1 \times \cdots \times \Omega_n, \mathcal{F}_1 \otimes \cdots \otimes \mathcal{F}_n)$ are given for each $n \in \mathbb{N}$.
    \begin{align*}
        & ^\forall n \in \mathbb{N} \hspace{0.5em} \text{and} \hspace{0.5em} A \in \mathcal{F}_1 \otimes \cdots \otimes \mathcal{F}_n, \hspace{0.8em} \mu_{n+1}(A\times \Omega_{n+1}) = \mu_n (A) \\
        &\implies \exists \text{ a probability space } (\Omega, \mathcal{F}, \mathbb{P}) \\
        &\text{ and random elements } X_1, X_2, \ldots \text{ with } X_t : \Omega \rightarrow \Omega_t \text{ such that } \mathbb{P}_{X_1, \ldots, X_n} = \mu_n \text{ for all } n.
    \end{align*}
\end{theorem*}

\begin{definition*}{}
    Let $(\mathcal{X}, \mathcal{F})$ and $(\mathcal{Y}, \mathcal{G})$ be measurable spaces. 
    A \textbf{probability kernel} or \textbf{Markov kernel} form $(\mathcal{X}, \mathcal{F})$ to $(\mathcal{Y}, \mathcal{G})$ is a function $K : \mathcal{X} \times \mathcal{G} \rightarrow [0,1]$ such that
    \begin{itemize}
        \item[(a)] $K(x,\cdot)$ is a measure for all $x \in \mathcal{X}$ and
        \item[(b)] $K(\cdot,A)$ is $\mathcal{F}$-measurable for all $A \in \mathcal{G}$. 
    \end{itemize}
    Notation: $K_x(A)$ or $K(A|x)$
\end{definition*}

\begin{definition*}{}
    Let $K_1$ be a probability kernel from $(\mathcal{X}, \mathcal{F}) \rightarrow (\mathcal{Y},\mathcal{G})$ and $K_2$ be a probability kernel from $(\mathcal{Y}, \mathcal{G}) \rightarrow (\mathcal{Z},\mathcal{H})$.
    Then the product kernel $K_1 \otimes K_2$ is the probability kernel from $(\mathcal{X}, \mathcal{F})$ to $(\mathcal{Y} \times \mathcal{Z}, \mathcal{G} \otimes \mathcal{H})$ defined by
    \begin{equation*}
        (K_1 \otimes K_2)(x,A)= \int_{\mathcal{Y}}\int_{\mathcal{Z}}\mathbbm{1}_A ((y,z))K_2 (y,dz) K_1 (x,dy).
    \end{equation*}
\end{definition*}

\begin{theorem*}{3.3}
    Let $(\Omega_n, \mathcal{F}_n)_{n=1}^\infty$ be a sequence of measurable spaces.
    \begin{align*}
    & K_1 \text{ : a probability measure on } (\Omega_1, \mathcal{F}_1) \text{, } \\
    & K_n \text{ : a probability kernel from } \Pi_{t=1}^{n-1} \Omega_t \text{ to } \Omega_n \text{ for } n \geq 2. \\
    & \implies \exists \text{ a probability space } (\Omega, \mathcal{F}, \mathbb{P}) \\
    & \text{ and random elements } (X_t)_{t=1}^\infty \text{ with } X_t : \Omega \rightarrow \Omega_t \text{ such that } \mathbb{P}_{X_1, \ldots, X_n} = \bigotimes_{t=1}^n K_t \text{ for all } n.
    \end{align*} 
\end{theorem*}

\begin{definition*}{}
    A \textbf{homogeneous Markov chain} is a sequence of random elements $(X_t)_{t=1}^\infty$ taking values in \textbf{state space} $\mathcal{S} = (\mathcal{X},\mathcal{F})$ and with
    \begin{equation*}
        \mathbb{P}(X_{t+1} \in \cdot \hspace{0.2em} | X_1, \ldots, X_t) = \mathbb{P}(X_{t+1} \in \cdot \hspace{0.2em} | X_t) = \mu(X_t, \cdot), \hspace{1.5em} \text{a.s.}
    \end{equation*}
    Here $\mu$ is a probability kernel from $(\mathcal{X}, \mathcal{F})$ to $(\mathcal{X}, \mathcal{F})$.
\end{definition*}

\begin{definition*}{3.4}
    A $\mathbb{F}$-adapted sequence of random variables $(X_t)_{t \in \mathbb{N}_+}$(i.e. $X_t$ is $\mathcal{F}_t$-measurable for all $t \in \mathbb{N}$) is a $\mathbb{F}$-adapted \textbf{martingale} if
    \begin{itemize}
        \item[(a)] $\mathbb{E}[|X_t|] < \infty \text{ for all } t.$
        \item[(b)] $\mathbb{E}[X_{t+1} | \mathcal{F}_t] = X_t \text{ for all } t.$
    \end{itemize}
    Supermartingale if less-than, submartingale if greater-than.
\end{definition*}

\begin{example*}{3.5}
    $(Y_t)_{t \in \mathbb{N}}$ be a sequence of independent random variables with $\mathbb{P}(Y_t = 1) = \mathbb{P}(Y_t = -1) = \frac{1}{2}$.
    Then the winnings after $t$ rounds is $S_t = \sum_{i=1}^{t} Y_i$, a martingale adapted to the filtration $(\mathcal{F}_t)_{t \in \mathbb{N}_+}$, where $\mathcal{F}_t = \sigma(Y_1, \ldots, Y_t)$.
\end{example*}

\begin{definition*}{3.6}
    Let $\mathbb{F} = (\mathcal{F}_t)_{t \in \mathbb{N}}$ be a filtration.
    A random variable $\tau \in \mathbb{N} \cup \{\infty\}$ is a \textbf{stopping time} with respect to the filtration $\mathbb{F}$ if
    \begin{equation*}
        \{\tau \leq t\} \in \mathcal{F}_t \text{ for all } t \in \mathbb{N}.
    \end{equation*}
    The $\sigma$-algebra generated by the stopping time $\tau$ is
    \begin{equation*}
        \mathcal{F}_\tau = \{A \in \mathcal{F}_\infty : A \cap \{\tau \leq t\} \in \mathcal{F}_t \text{ for all } t \in \mathbb{N} \}.
    \end{equation*}
\end{definition*}

\begin{theorem*}{3.8}[Doob's Optional Stopping Theorem]
    Let $\mathbb{F} = (\mathcal{F}_t)_{t \in \mathbb{N}}$ be a filtration and $(X_t)_{t \in \mathbb{N}}$ be an $\mathbb{F}$-adapted martingale.
    Let $\tau$ be a stopping time with respect to $\mathbb{F}$,
    and one of the following conditions is satisfied:
    \begin{itemize}
        \item[(a)] There exists an $n \in \mathbb{N}$ such that $\mathbb{P}(\tau > n) = 0$.
        \item[(b)] $\mathbb{E}[\tau] < \infty$ and there exists a constant $c \in \mathbb{R}$ such that $|X_{t+1} - X_t| \leq c$ for all $t$, $\mathbb{E}[|X_{t+1} - X_t| | \mathcal{F}_t] \leq c$ a.s. for $\tau > t$.
        \item[(c)] There exists a constant $c$ such that $|X_{t \wedge \tau}| \leq c$ a.s. for all $t \in \mathbb{N}$.
    \end{itemize}
    Then $X_\tau$ is almost surely well-defined and 
    \begin{equation*}
        \mathbb{E}[X_\tau] = \mathbb{E}[X_0].
    \end{equation*}
\end{theorem*}

\begin{theorem*}{3.9}[Maximal inequality]
    Let $(X_t)_{t=0}^\infty$ be a supermartingale with $X_t \geq 0$ a.s. for all $t \in \mathbb{N}$.
    Then for any $\epsilon > 0$,
    \begin{equation*}
        \mathbb{P} \left( \sup_{t\in\mathbb{N}} X_t \geq \epsilon \right) \leq \frac{\mathbb{E}[X_0]}{\epsilon}.
    \end{equation*}
\end{theorem*}

\begin{theorem*}{3.10}
    Let $(X_t)_{t=0}^\infty$ be a submartingale with $X_t \geq 0$ a.s. for all $t \in \mathbb{N}$.
    Then for any $\epsilon > 0$,
    \begin{equation*}
        \mathbb{P} \left( \max_{t\in {0,1,\ldots,n}} X_t \geq \epsilon \right) \leq \frac{\mathbb{E}[X_n]}{\epsilon}.
    \end{equation*}

\end{theorem*}

\subsection{Stochastic Bandits}
\begin{definition*}{}
    A \textbf{stochastic bandit} is a collection of distributions $\nu = \{P_a : a \in \mathcal{A} \}$.
\end{definition*}
Here we are challenging to maximize the total reward $S_n = \sum_{t=1}^{n} X_t$ over $n$ rounds.
\begin{definition*}{}
    A set of bandits $\mathcal{E}$ for which $\nu \in \mathcal{E}$ is called the environment class.
    \begin{itemize}{}
        \item An environment class $\mathcal{E}$ is \textbf{unstructured} if $\mathcal{A}$ is finite and
        \begin{equation*}
            ^\forall \hspace{0.1em} a \in \mathcal{A}, \hspace{0.2em} \exists \text{ sets of distributions } \mathcal{M}_a \text{ such that } \mathcal{E} = \{ \nu = (P_a : a \in \mathcal{A}) : P_a \in \mathcal{M}_a, \hspace{0.2em} ^\forall \hspace{0.1em} a \in \mathcal{A}\}.
        \end{equation*}
        or, in short, $\mathcal{E} = \times_{a \in \mathcal{A}} \mathcal{M}_a$.
        \item Environment classes with finite number of degrees of freedom are called \textbf{parametric} (Bernoulli).
        \item Environment classes that are not parametric are called \textbf{nonparametric} (subgaussian).
        \item An environment class is called \textbf{structured} if it is not unstructured.
    \end{itemize}
\end{definition*}

\begin{example*}{4.1}
    Let $\mathcal{A} = \{1, 2\}$ and $\mathcal{E} = \{ (\mathcal{B}(\theta), \mathcal{B}(1-\theta)): \theta \in [0,1]\}$ where $\mathcal{B}(\theta)$ denotes the Bernoulli distribution with mean $\theta$.
    Then $\mathcal{E}$ is a structured parametric environment class.
\end{example*}

\begin{example*}{4.2} [Stochastic linear bandit]
    Let $\mathcal{A} \subseteq \mathbb{R}^d$ and $\theta \in \mathbb{R}^d$ and
    \begin{equation*}
        \nu_\theta = ( \mathcal{N}(\langle a, \theta \rangle, 1) : a \in \mathcal{A} ) \hspace{2em} \text{ and } \hspace{0.3em} \mathcal{E} = \{ \nu_\theta : \theta \in \mathbb{R}^d \}.
    \end{equation*}
\end{example*}

\begin{example*}{4.3}
    Consider an undirected graph $G$ with vertices $V={1, \ldots, |V|}$ and edges $E={1, \ldots, |E|}$.
    In each round the learner chooses a path from vertex 1 to vertex $|V|$.
    Then each edge $e \in [E]$ is removed from the graph with probability $1-\theta_e$ for unknown $\theta \in [0,1]^{|E|}$.
    In this problem, $\mathcal{A}$ is the set of paths and
    \begin{equation*}
        \nu_\theta = \left( \mathcal{B}\left(\prod_{e \in a} \theta_e\right) : a \in \mathcal{A} \right) \hspace{2em} \text{ and } \hspace{0.3em} \mathcal{E} = \{ \nu_\theta : \theta \in [0,1]^{|E|} \}.
    \end{equation*}
\end{example*}

\begin{definition*}{}
    Let $\nu = (P_a : a \in \mathcal{A})$ be a stochastic bandit.
    \begin{equation*}
        \mu_a(\nu) := \int_{-\infty}^{\infty} x \, dP_a(x)
    \end{equation*}
    Also let $\mu^*(\nu) := \max_{a \in \mathcal{A}} \mu_a(\nu)$, the largest mean of all the arms.
\end{definition*}

\begin{definition*}{}
    The \textbf{regret} of policy $\pi$ on bandit instance $\nu$ is
    \begin{equation*}
        R_n (\pi, \nu) := n \mu^*(\nu) - \mathbb{E}\left[\sum_{t=1}^n X_t\right]
    \end{equation*}
    where the expectation is taken with respect to the probability measure on outcomes induced by the interaction of $\pi$ and $\nu$.
\end{definition*}

\begin{lemma*}{4.4}
    Let $\nu$ be a stochastic bandit environment. Then
    \begin{itemize}
        \item $R_n(\pi, \nu) \geq 0$ for all policies $\pi$,
        \item the policy $\pi$ choosing $A_t \in \arg\max_a \mu_a(\nu)$ for all $t$ satisfies $R_n(\pi, \nu) = 0$, and
        \item if $R_n(\pi, \nu) = 0$ for some policy $\pi$, then $\mathbb{P} \left[\mu_{A_t}(\nu) = \mu^*(\nu) \right] = 1$ for all $t \in [n]$.
    \end{itemize}
\end{lemma*}

\begin{proof}
    (a) Since $\mu^*(\nu) = \max_{a \in \mathcal{A}} \mu_a(\nu)$,
    \begin{equation*}
        n \mu^*(\nu) \geq \mathbb{E} \left[ \sum_{t=1}^n \mu_{A_t}(\nu) \right].
    \end{equation*}
    By the linearity of expectation,
    \begin{equation*}
        \mathbb{E} \left[ \sum_{t=1}^n \mu_{A_t}(\nu) \right] = \sum_{t=1}^n \mathbb{E}\left[\mu_{A_t}(\nu)\right].
    \end{equation*}
    By the stochastic bandit assumption and the tower property,
    \begin{align*}
        \sum_{t=1}^n \mathbb{E}\left[\mu_{A_t}(\nu)\right] &= \sum_{t=1}^n \mathbb{E}\left[\mathbb{E}\left[X_t \mid H_{t-1}, A_t\right]\right] \\
        &= \sum_{t=1}^n \mathbb{E}\left[X_t\right] \\
        &= \mathbb{E}\left[\sum_{t=1}^n X_t\right],
    \end{align*}
    using linearity of expectation again. Therefore,
    \begin{equation*}
        R_n(\pi, \nu) = n \mu^*(\nu) - \mathbb{E}\left[\sum_{t=1}^n X_t\right] \geq 0.
    \end{equation*}

    (b) Let $\pi$ be the policy choosing $A_t \in \arg\max_a \mu_a(\nu)$ for all $t$. Then $\mu^*(\nu) = \mu_{A_t}(\nu)$ for all $t$, which implies
    \begin{align*}
        R_n(\pi, \nu) &= n \mu^*(\nu) - \mathbb{E}\left[\sum_{t=1}^n X_t\right] \\
        &= \sum_{t=1}^n \left(\mu^*(\nu) - \mathbb{E}\left[\mu_{A_t}(\nu)\right]\right) \\
        &= \mathbb{E}\left[\sum_{t=1}^n \left( \mu^*(\nu) - \mu_{A_t}(\nu)\right) \right] \\
        &= 0
    \end{align*}
    and therefore $R_n(\pi, \nu) = 0$.

    (c) Suppose $R_n(\pi, \nu) = 0$ for some policy $\pi$. Then
    \begin{equation*}
        \sum_{t=1}^n \mathbb{E}\left[\mu^*(\nu) - \mu_{A_t}(\nu)\right] = 0.
    \end{equation*}
    Since $\mu^*(\nu) - \mu_{A_t}(\nu) \geq 0$ for all $t$, $\mu^*(\nu) = \mu_{A_t}(\nu)$ almost surely for all $t \in \{1, \ldots, n\}$.

\end{proof}

\begin{remark*}{}[Factorisation of the regret]
    The learner cannot always achieve zero regret, so the relatively weak objective is to find a policy $\pi$ with
    \begin{equation*}
        ^\forall \hspace{0.2em} \nu \in \mathcal{E}, \hspace{1em} \lim_{n \to \infty} \frac{R_n(\pi, \nu)}{n} = 0.
    \end{equation*}
    Or further objective, finding some constants $C > 0$ and $p < 1$ such that
    \begin{equation*} 
        ^\forall \hspace{0.2em} \nu \in \mathcal{E}, \hspace{1em} R_n(\pi, \nu) \leq C n^p.
    \end{equation*}
    Another alternative is to find a function $C: \mathcal{E} \to [0, \infty)$ and $f : \mathbb{N} \to [0, \infty)$ such that
    \begin{equation*}
        ^\forall \hspace{0.2em} \nu \in \mathcal{E}, \hspace{1em} R_n(\pi, \nu) \leq C(\nu) f(n).
    \end{equation*}
    \textbf{Bayesian regret} is another candidate objective. For prior probability measure $Q$ on $\mathcal{E}$, 
    the Bayesian regret is the average of the regret with respect to the prior $Q$, i.e.,
    \begin{equation*}
        \text{BR}_n(\pi, Q) = \int_{\mathcal{E}} R_n(\pi, \nu) \, dQ(\nu)
    \end{equation*}
    where the regret $R_n$ is measurable.
\end{remark*}

\begin{definition*}{}
    Let $\nu = \left( P_a : a \in \mathcal{A} \right)$ be a stochastic bandit.
    Then the \textbf{suboptimality gap} of action a is defined by
    \begin{equation*}
        \Delta_a(\nu) := \mu^*(\nu) - \mu_a(\nu).
    \end{equation*}
    Also let
    \begin{equation*}
        T_a(t) := \sum_{s=1}^t \mathbbm{1}\{A_s = a\}
    \end{equation*}
    be the number of times action $a$ has been chosen up to time $t$.
\end{definition*}

\begin{lemma*}{4.5}[Regret decomposition lemma]
    For any policy $\pi$ and stochastic bandit $\nu$ with $\mathcal{A}$ finite or countable and horizon $n \in \mathbb{N}$, the regret $R_n$ of policy $\pi$ in $\nu$ satisfies
    \begin{equation*}
        R_n(\pi, \nu) = \sum_{a \in \mathcal{A}} \Delta_a(\nu) \mathbb{E}[T_a(n)].
    \end{equation*}
\end{lemma*}

\begin{proof}
    For any $t \in \mathbb{N}$, $\sum_{a \in \mathcal{A}} \mathbbm{1}\{A_t = a\} = 1$. Hence $S_n = \sum_t X_t = \sum_t \sum_a X_t \mathbbm{1}\{A_t = a\}$.
    That is
    \begin{equation*}
        R_n = n \mu^* - \mathbb{E} [S_n] = \sum_{a \in \mathcal{A}} \sum_{t=1}^n \mathbb{E} [( \mu^* - X_t) \mathbbm{1}\{ A_t = a \}].
    \end{equation*}
    Since $\mathbb{E} [X_t \mid H_{t-1}, A_t] = \mu_{A_t}$, we have
    \begin{align*}
        \mathbb{E}[(\mu^* - X_t) \mathbbm{1}\{A_t = a\} \mid A_t] &= \mathbbm{1}\{A_t = a\} \mathbb{E}[\mu^* - X_t \mid A_t] \\
        &= \mathbbm{1}\{A_t = a\} (\mu^* - \mu_a) \\
        &= \mathbbm{1}\{A_t = a\} \Delta_a.
    \end{align*}
    Therefore
    \begin{align*}
        R_n &= \sum_{a \in \mathcal{A}} \sum_{t=1}^n \mathbb{E}[(\mu^* - X_t) \mathbbm{1}\{A_t = a\}] \\
            &= \sum_{a \in \mathcal{A}} \sum_{t=1}^n \mathbb{E}[\mathbb{E}[(\mu^* - X_t) \mathbbm{1}\{A_t = a\} \mid A_t]] \\
            &= \sum_{a \in \mathcal{A}} \sum_{t=1}^n \mathbb{E}[\Delta_a \mathbbm{1}\{A_t = a\}] \\
            &= \sum_{a \in \mathcal{A}} \Delta_a \mathbb{E}[T_a(n)].
    \end{align*}
\end{proof}

\begin{theorem*}{}[Cramér]
    Suppose $X_1, \ldots, X_n$ are i.i.d. random variables whose values lie in $\mathbb{R}^d$ with finite empirical mean defined by $S_n = \frac{1}{n} \sum_{i=1}^n X_i$.
    Define $\mu_n$ to be the law of $S_n$.
    Then $\mu_n$ satisfies a large deviation principle on $\mathbb{R}^d$ with rate function given by the Legendre transform of the cumulant generating function of $X_1$.
    That is,
    \begin{itemize}
        \item For any closed set $F \subset \mathbb{R}^d$,
        \begin{equation*}
            \limsup_{n \to \infty} \frac{1}{n} \log \mu_n(F) \leq -\inf_{x \in F} I(x),
        \end{equation*}
        \item For any open set $G \subset \mathbb{R}^d$,
        \begin{equation*}
            \liminf_{n \to \infty} \frac{1}{n} \log \mu_n(G) \geq -\inf_{x \in G} I(x),
        \end{equation*}
    where $I(x)$ is the rate function given by the Legendre transform of the cumulant generating function of $X_1$.
    \end{itemize}
\end{theorem*}

\begin{theorem*}{}[Gärtner-Ellis]
    Suppose $X_1, \ldots, X_n$ are random variables whose values lie in $\mathbb{R}^d$ with finite empirical mean defined by $S_n = \frac{1}{n} \sum_{i=1}^n X_i$.
    Define $\mu_n$ to be the law of $S_n$.
    Suppose the limit
    \begin{equation*}
        \Lambda(\lambda) := \lim_{n \to \infty} \frac{1}{n} \log \mathbb{E}\left[e^{n \langle \lambda, S_n \rangle}\right]
    \end{equation*}
    exists for all $\lambda \in \mathbb{R}^d$ and $0 \in \mathcal{D}_\Lambda^\circ$, where $\mathcal{D}_\Lambda$ is the effective domain of $\Lambda$.
    Then
    \begin{itemize}
        \item For any closed set $F \subset \mathbb{R}^d$,
        \begin{equation*}
            \limsup_{n \to \infty} \frac{1}{n} \log \mu_n(F) \leq -\inf_{x \in F} I(x),
        \end{equation*}
        \item For any open set $G \subset \mathbb{R}^d$,
        \begin{equation*}
            \liminf_{n \to \infty} \frac{1}{n} \log \mu_n(G) \geq -\inf_{x \in G \cap \mathcal{F}} \Lambda^*(x),
        \end{equation*}
    where $\Lambda^*(x)$ is the rate function given by the Legendre transform of $\Lambda(\lambda)$, and $\mathcal{F}$ is the set of exposed points of $\Lambda^*(x)$.
    \end{itemize}
    Furthermore, if $\Lambda$ is essentially smooth and lower semicontinuous, then $\mathcal{F}$ can be replaced by the entire domain of $\Lambda^*$.(LDP holds)
\end{theorem*}
